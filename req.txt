    @classmethod
    def parse_orm(cls, orm_result: database.Configuration, session: Session) -> "DQConfigDetails":
        has_ever_succeeded = False
        release_to_prod_id = None

        for status in orm_result.job_statuses:
            if status.status == "SUCCESS":
                has_ever_succeeded = True
                break

        if has_ever_succeeded:
            job_history_id = orm_result.id
        else:
            job_history_id = None

        cosmos_uid = None
        log_uid = None
        end_timestamp = None
        result = None
        status = "Not Started"
        has_batch = False
        
        # NEW: Initialize optional timing fields
        start_timestamp = None
        job_run_time = None

        for job in orm_result.jobs:
            if job.batch:
                has_batch = True
                break

        if orm_result.job_statuses:
            last_job = sorted(
                list(orm_result.job_statuses), key=lambda js: js.created_date
            )[-1]
            log_uid = last_job.name
            status = READABLE_STATUSES.get(last_job.status, last_job.status)
            
            # SAFELY get job timing information (handle missing attributes)
            start_timestamp = getattr(last_job, 'start_timestamp', None)
            end_timestamp = getattr(last_job, 'end_timestamp', None)
            
            # SAFELY calculate job run time if we have start_timestamp
            if start_timestamp:
                try:
                    # Determine if job is completed
                    completed_statuses = ['SUCCESS', 'FAILED', 'CANCELLED', 'COMPLETED']
                    is_completed = last_job.status.upper() in completed_statuses
                    
                    # Calculate duration
                    if is_completed and end_timestamp:
                        # Job completed - use end_timestamp
                        duration = end_timestamp - start_timestamp
                    else:
                        # Job is still running - calculate up to now
                        duration = datetime.datetime.utcnow() - start_timestamp
                    
                    # Format as HH:MM:SS
                    total_seconds = int(duration.total_seconds())
                    hours = total_seconds // 3600
                    minutes = (total_seconds % 3600) // 60
                    seconds = total_seconds % 60
                    job_run_time = f"{hours:02d}:{minutes:02d}:{seconds:02d}"
                    
                except (AttributeError, TypeError, ValueError):
                    # Gracefully handle any calculation errors
                    job_run_time = None

            if last_job.status == "SUCCESS":
                cosmos_uid = last_job.name
                end_timestamp = getattr(last_job, 'end_timestamp', None)
                # Get execution result for completed jobs
                try:
                    result = get_execution_result(last_job.results_and_breaches)
                except (AttributeError, TypeError):
                    result = None
        else:
            if not orm_result.released_to_prod:
                release_to_prod_id = orm_result.id
            else:
                successful_jobs = [
                    j for j in orm_result.job_statuses if j.status == "SUCCESS"
                ]
                if successful_jobs:
                    latest_success = sorted(
                        list(successful_jobs), key=lambda js: js.created_date
                    )[-1]
                    end_timestamp = getattr(latest_success, 'end_timestamp', None)
                    
                    # SAFELY get timing for successful jobs
                    start_timestamp = getattr(latest_success, 'start_timestamp', None)
                    if start_timestamp and end_timestamp:
                        try:
                            duration = end_timestamp - start_timestamp
                            total_seconds = int(duration.total_seconds())
                            hours = total_seconds // 3600
                            minutes = (total_seconds % 3600) // 60
                            seconds = total_seconds % 60
                            job_run_time = f"{hours:02d}:{minutes:02d}:{seconds:02d}"
                        except (AttributeError, TypeError, ValueError):
                            job_run_time = None

        # Get modified details
        try:
            modified_details = cls.get_modified_by_config_id(cls, session, orm_result.id)
        except Exception:
            modified_details = {"modified_by": "Anonymous", "modified_at": None}
        
        # Prepare result dictionary
        orm_result_dict = {
            "id": orm_result.id,
            "file_name": orm_result.file_name,
        }

        return cls(
            cosmos_uid=cosmos_uid,
            log_uid=log_uid,
            job_history_id=job_history_id,
            status=status,
            release_to_prod_id=release_to_prod_id,
            end_timestamp=end_timestamp,
            result=result,
            uploader_name=orm_result.uploader_name,
            modified_by=orm_result.uploader_name
            if str(modified_details.get("modified_by", "Anonymous")).lower() == "anonymous"
            else modified_details.get("modified_by", "Anonymous"),
            modified_at=orm_result.upload_datetime
            if not modified_details.get("modified_at")
            else modified_details.get("modified_at"),
            has_batch=has_batch,
            # SAFELY ADDED OPTIONAL FIELDS
            start_timestamp=start_timestamp,
            job_run_time=job_run_time,
            **orm_result_dict
        )

    @classmethod
    def load_by_name(cls, session: Session, file_name: str, team_id: int) -> "DQConfigDetails":
        result = (
            session.query(
                database.Configuration,
            )
            .where(
                database.Configuration.active,
                database.Configuration.file_name == file_name,
                database.Configuration.team_id == team_id,
            )
            .order_by(
                database.Configuration.upload_datetime.desc(),
            )
            .options(joinedload(database.Configuration.job_statuses))
            .first()
        )

        if result:
            return cls.parse_orm(result, session)
        else:
            raise DoesNotExistsException("Configuration not found")

    def get_modified_by_config_id(self, session: Session, dq_config_id):
        stmt = (
            union_all(
                select(
                    database.Rule.modified_by.label("modified_by"),
                    database.Rule.modified_at.label("modified_at"),
                )
                .join(database.Batch, database.Rule.batch_id == database.Batch.id)
                .join(database.Job, database.Job.batch_id == database.Batch.id)
                .where(
                    database.Job.configuration_id == dq_config_id,
                    database.Job.actv_flag_ind == Active.Y,
                    database.Batch.soft_delete == False,
                    database.Rule.soft_delete.is_(None),
                    database.Rule.modified_at.isnot(None),
                ),
                select(
                    database.Batch.modified_by.label("modified_by"),
                    database.Batch.modified_at.label("modified_at"),
                )
                .join(database.Job, database.Job.batch_id == database.Batch.id)
                .where(
                    database.Job.configuration_id == dq_config_id,
                    database.Job.actv_flag_ind == Active.Y,
                    database.Batch.soft_delete == False,
                    database.Batch.modified_at.isnot(None),
                ),
            )
            .order_by(text("modified_at DESC"))
            .limit(1)
        )
        data = session.execute(stmt).first()
        if data:
            return {"modified_by": data.modified_by, "modified_at": data.modified_at}
        return {"modified_by": "Anonymous", "modified_at": None}


class DQConfigDetailsList(PaginatedModel):
    data: List[DQConfigDetails]

    @classmethod
    def load(
        cls, session: Session, filters: DQConfigDetailsFilters
    ) -> Union["DQConfigDetailsList", Table]:
        where_conditions = [
            database.Configuration.team_id == filters.team_id,
            database.Configuration.active,
        ]

        query = session.query(
            database.Configuration,
        ).where(
            *where_conditions,
        )

        count = query.count()

        results = (
            query.order_by(
                filters.sorting_key.desc(),
            )
            .offset(
                filters.page_number * filters.items_per_page,
            )
            .limit(
                filters.items_per_page,
            )
            .options(joinedload(database.Configuration.job_statuses))
            .all()
        )

        entities = []

        for result in results:
            entities.append(DQConfigDetails.parse_orm(result, session))

        if filters.as_table:
            # Use original TABLE_MAPPING (without job_run_time) for backward compatibility
            return Table.convert(
                entities,
                TABLE_MAPPING,
                page_number=filters.page_number,
                items_per_page=filters.items_per_page,
                total_number_of_items=count,
            )

        return cls(
            data=entities,
            page_number=filters.page_number,
            items_per_page=filters.items_per_page,
            total_number_of_items=count,
        )

def validate_base_population(data: Any):
    try:
        base_population = getattr(data, "base_pop", "")
        if not base_population or len(base_population.split()) < 2:
            raise ValueError("base_pop not a valid query provided")
        database_type = str(data.database_type.value).replace(" ", "").lower()
        sqlglot.parse_one(
            base_population,
            dialect="duckdb" if database_type == "file" else database_type,
        )
        return data
    except ParseError as e:
        raise ValueError(f"base_pop not a valid query. exception occurred : {e}")
